<script>
    function calculateScore() {
        const transport = parseInt(document.getElementById('transport').value);
        const school = parseInt(document.getElementById('school').value);
        const infra = parseInt(document.getElementById('infra').value);
        const brand = parseInt(document.getElementById('brand').value);
        const floor = parseInt(document.getElementById('floor').value);
        const age = parseInt(document.getElementById('age').value);

        const totalScore = transport + school + infra + brand + floor + age;

        let grade = '';
        let color = '';

        if (totalScore >= 90) {
            grade = '🌟 S등급 (매우 우수한 입지 및 상품성)';
            color = '#28a745';
        } else if (totalScore >= 75) {
            grade = '👍 A등급 (우수한 입지/실거주 및 투자 양호)';
            color = '#007bff';
        } else if (totalScore >= 60) {
            grade = '👌 B등급 (보통 수준의 아파트)';
            color = '#ffc107';
        } else {
            grade = '⚠️ C등급 (입지 요소나 상품성 보완 필요)';
            color = '#dc3545';
        }

        document.getElementById('scoreDisplay').innerText = totalScore + '점 / 100점';
        document.getElementById('gradeDisplay').innerText = grade;
        document.getElementById('gradeDisplay').style.color = color;
        document.getElementById('resultBox').style.display = 'block';
    }
</script>
